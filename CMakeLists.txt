cmake_minimum_required(VERSION 3.16)
project(TheLastWatch VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(ttrpg_engine entity.cpp player.cpp enemy.cpp items.cpp map.cpp boss.cpp game.cpp campaign_io.cpp)
target_include_directories(ttrpg_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
    target_compile_options(ttrpg_engine PUBLIC /W4)
else()
    target_compile_options(ttrpg_engine PUBLIC -Wall -Wextra -Wpedantic)
    if(ENABLE_SANITIZERS)
        target_compile_options(ttrpg_engine PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(ttrpg_engine PUBLIC -fsanitize=address,undefined)
    endif()
endif()
add_executable(ttrpg main.cpp)
target_link_libraries(ttrpg PRIVATE ttrpg_engine)
option(BUILD_GUI "Build the SDL2 desktop interface" ON)
if(BUILD_GUI)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    find_package(Freetype REQUIRED)
    add_executable(ttrpg_gui gui/main.cpp gui/app.cpp gui/draw.cpp gui/icons.cpp gui/presentation.cpp gui/tutorial.cpp)
    target_link_libraries(ttrpg_gui PRIVATE ttrpg_engine PkgConfig::SDL2 Freetype::Freetype)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(engine_tests tests/engine_tests.cpp)
    target_link_libraries(engine_tests PRIVATE ttrpg_engine)
    add_test(NAME engine_rules COMMAND engine_tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME terminal_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:ttrpg>)
        if(BUILD_GUI)
            add_test(NAME graphical_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/gui_test.py $<TARGET_FILE:ttrpg_gui>)
        endif()
    endif()
endif()
